from typing import Optional
import hashlib

def secure_hash(data: str, salt: Optional[str] = None) -> str:
    """Generate a deterministic secure hash."""
    payload = f"{salt or ''}{data}".encode()
    return hashlib.sha256(payload).hexdigest()
